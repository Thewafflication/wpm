\def\TCID{TC-0001}

\def\TCTitle{Usage/version check}

\def\TCPurpose{
	Verify that wpm executes, exits cleanly, and reports its bundled and Windows system-library dependency versions.
}

\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{REQ-0001}

\def\TCDesignRef{Not applicable; this test verifies the public command-line interface.}
\def\TCFeatureRef{\texttt{wpm/main.c} and \texttt{tests/tc-0001-usage-version-check.ps1}}
\def\TCTestTechnique{Use-case testing with equivalence partitions for no-argument and version-query invocation.}
\def\TCAssumptions{The executable was produced from the source revision under test.}
\def\TCInitialState{No WPM process from an earlier execution is running.}

\def\TCPreconditions{
	None.
}

\def\TCEnvironment{
	Local Windows developer workstation. Future target: Windows Container.
}

\def\TCInputData{
	No input required.
}

\def\TCProcedure{
	\par
	\begin{enumerate}
		\item Run \texttt{wpm.exe}
		\item Run \texttt{wpm.exe --version}
	\end{enumerate}
}

\def\TCExpectedResult{
	\par
	\begin{enumerate}
		\item Exit code 0 and version or usage information is displayed.
		\item Exit code 0 and dependency version information is displayed for miniz, libsodium, urlmon, and advapi32.
	\end{enumerate}
}

\def\TCPostConditions{
	Test environment remains operational.
}
\TCTemplate
