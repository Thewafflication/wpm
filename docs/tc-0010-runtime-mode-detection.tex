\def\TCID{TC-0010}
\def\TCTitle{Runtime mode detection}
\def\TCPurpose{Verify managed and portable runtime detection plus portable execution from a read-only executable directory.}
\def\TCPriority{\TCPriorityHigh}
\def\TCRequirementRef{REQ-0010}
\def\TCDesignRef{Not applicable; runtime-mode behavior is controlled directly by REQ-0010.}
\def\TCFeatureRef{\texttt{wpm/main.c} and \texttt{tests/tc-0010-runtime-mode-detection.ps1}}
\def\TCTestTechnique{Decision-table testing of managed and portable locations combined with filesystem permission states.}
\def\TCAssumptions{The test account can create the simulated managed and portable directory structures.}
\def\TCInitialState{No WPM data directories exist in the isolated test roots.}
\def\TCPreconditions{A built WPM executable is available and temporary test directories are writable.}
\def\TCEnvironment{Local Windows developer workstation. Future target: Windows Container.}
\def\TCInputData{The built WPM executable, copied to simulated managed and portable locations.}
\def\TCProcedure{\par\begin{enumerate}\item Run a portable copy with \texttt{--verbose}.\item Run a copy beneath a simulated Program Files \texttt{WPM} directory with \texttt{--verbose}.\item Deny writes to the portable executable directory, then run \texttt{wpm.exe repo list}.\end{enumerate}}
\def\TCExpectedResult{\par\begin{enumerate}\item The portable copy reports portable mode and its executable path.\item The managed copy reports managed mode.\item The read-only portable copy creates the data, package, temporary, cache, and configuration directories without writing beside the executable.\end{enumerate}}
\def\TCPostConditions{Temporary executable and data directories are removed.}
